{{- /* Compact stories tray (home + /now). Params: count (default 4). Shows a "nothing in 24h" placeholder when none of the shown stories are fresh. */ -}} {{- $count := .count | default 4 }} {{- $stories := where site.RegularPages "Type" "in" (slice "stories") }} {{- $stories = sort $stories "Date" "desc" }} {{- $recent := first $count $stories }} {{- if $recent }} {{- $hasFresh := false }} {{- range $recent }} {{- $h := (now.Sub .Date).Hours }} {{- if and (ge $h 0.0) (lt $h 24.0) }}{{ $hasFresh = true }}{{ end }} {{- end }}
{{- if not $hasFresh }} {{- end }} {{- range $recent }} {{- $p := . }} {{- $images := .Resources.ByType "image" }} {{- with $images }} {{- $thumb := (index . 0).Fill "200x200 webp q80" }} {{- $alt := $p.Params.caption | default $p.Title | default ($p.Date.Format "Jan 2, 2006") }} {{- $h := (now.Sub $p.Date).Hours }} {{- $fresh := and (ge $h 0.0) (lt $h 24.0) }} {{- end }} {{- end }}
{{- end }}
